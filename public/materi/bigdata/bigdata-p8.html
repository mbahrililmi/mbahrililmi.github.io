Contoh Soal Praktikum

# SOAL PRAKTIKUM UTS - CONTOH
# Dataset: Iris dataset (akan diganti dengan dataset ujian yang sebenarnya)

import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.tree import DecisionTreeClassifier
from sklearn.cluster import KMeans
from sklearn.metrics import classification_report, silhouette_score

# === BAGIAN A: CLASSIFICATION (30 poin) ===
from sklearn.datasets import load_iris
iris = load_iris()
X = pd.DataFrame(iris.data, columns=iris.feature_names)
y = iris.target

# 1. Split data (80:20, stratified)
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42
)

# 2. Train Decision Tree
dt = DecisionTreeClassifier(max_depth=4, random_state=42)
dt.fit(X_train, y_train)
y_pred = dt.predict(X_test)

# 3. Evaluate - WAJIB interpretasikan setiap metrics!
print("=== Classification Report ===")
print(classification_report(y_test, y_pred, target_names=iris.target_names))

# === BAGIAN B: CLUSTERING (30 poin) ===
# 4. Standardize features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 5. K-Means dengan K optimal (tentukan dari Elbow Method)
kmeans = KMeans(n_clusters=3, init='k-means++', random_state=42)
labels = kmeans.fit_predict(X_scaled)

# 6. Evaluate clustering
score = silhouette_score(X_scaled, labels)
print(f"
Silhouette Score: {score:.3f}")
print("Interpretasi: ...[tulis interpretasi bisnis di sini]")
